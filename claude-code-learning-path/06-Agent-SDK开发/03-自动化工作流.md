# 自动化工作流

## Headless 模式

Headless 模式（非交互模式）是 Claude Code 在 CI/CD 和自动化场景中的核心用法。

### 单次查询

```bash
# 最简单的 headless 调用
claude -p "解释项目架构" --output-format json

# 指定工具范围
claude -p "运行测试并修复失败的用例" \
  --allowedTools "Read,Bash(npm test:*),Edit" \
  --output-format json \
  --max-turns 30
```

### 从代码调用

```python
import subprocess
import json


def claude_headless(prompt: str, max_turns: int = 30) -> dict:
    """调用 Claude Code headless 模式"""
    result = subprocess.run(
        [
            "claude",
            "-p", prompt,
            "--output-format", "json",
            "--max-turns", str(max_turns),
            "--allowedTools", "Read,Write,Edit,Bash(git *),Bash(npm *)"
        ],
        capture_output=True,
        text=True,
        timeout=600  # 10 分钟超时
    )
    return json.loads(result.stdout)


# 使用
result = claude_headless("为 src/utils.ts 添加单元测试")
print(f"状态: {result['status']}")
print(f"Token 消耗: {result['total_tokens']}")
```

## CI/CD 集成

### GitHub Actions

```yaml
# .github/workflows/claude-review.yml
name: Claude Code Review

on:
  pull_request:
    types: [opened, synchronize]

jobs:
  review:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '20'

      - name: Install Claude Code
        run: npm install -g @anthropic-ai/claude-code

      - name: Claude Code Review
        env:
          ANTHROPIC_API_KEY: ${{ secrets.ANTHROPIC_API_KEY }}
        run: |
          claude -p "Review the code changes in this PR. \
            Check for bugs, security issues, and code quality. \
            Output findings in a structured format." \
            --output-format json \
            --allowedTools "Read,Grep,Glob,Bash(git diff:*,git log:*)" \
            --max-turns 20 > review.json

      - name: Post Review
        uses: actions/github-script@v7
        with:
          script: |
            const review = require('./review.json');
            await github.rest.issues.createComment({
              ...context.repo,
              issue_number: context.issue.number,
              body: review.result
            });
```

## 批量任务处理

### 批量代码迁移

```python
import asyncio
from claude_agent_sdk import ClaudeAgentClient


async def migrate_file(file_path: str, migration_rules: str):
    """使用 Claude 迁移单个文件"""
    async with ClaudeAgentClient() as client:
        prompt = f"""Apply the following migration to {file_path}:
        {migration_rules}

        Rules:
        - Replace var with const/let
        - Convert callbacks to async/await
        - Add TypeScript type annotations
        """
        return await client.query(
            prompt,
            allowed_tools=["Read", "Edit"],
            permission_mode="acceptEdits"
        )


async def batch_migrate(files: list[str], rules: str):
    """并行迁移多个文件（每个在独立子进程中）"""
    tasks = [migrate_file(f, rules) for f in files]
    results = await asyncio.gather(*tasks, return_exceptions=True)

    for file_path, result in zip(files, results):
        if isinstance(result, Exception):
            print(f"FAILED: {file_path} - {result}")
        else:
            print(f"OK: {file_path}")
```

### 自动生成测试

```python
async def generate_tests(source_files: list[str]):
    """为指定文件自动生成单元测试"""
    async with ClaudeAgentClient() as client:
        for file_path in source_files:
            test_path = file_path.replace("src/", "tests/").replace(".ts", ".test.ts")

            async with client.session() as session:
                # 1. 读取源码
                await session.send(f"Read and understand the code in @{file_path}")

                # 2. 生成测试
                await session.send(f"""
                Generate comprehensive unit tests for @{file_path}.
                Write them to @{test_path}.

                Requirements:
                - Cover all exported functions
                - Include edge cases and error conditions
                - Use project's existing test framework
                - Follow project test conventions
                """)

                result = await session.receive()
                print(f"Generated tests: {test_path} ({result.total_tokens} tokens)")
```

## 多 Agent 编排

### 主从模式

```python
import asyncio
from claude_agent_sdk import ClaudeAgentClient


async def orchestrator(task: str):
    """主 Agent：分解任务，分配子任务"""
    async with ClaudeAgentClient() as client:
        # Step 1: 分析任务，生成子任务
        async with client.session() as session:
            await session.send(f"""
            Decompose this task into independent subtasks:
            "{task}"

            Output a JSON list: [{{"id": 1, "title": "...", "description": "..."}}]
            """)
            plan = await session.receive()

    return plan


async def worker(subtask: dict):
    """工作 Agent：执行具体子任务"""
    async with ClaudeAgentClient() as client:
        return await client.query(
            f"Execute this subtask: {subtask['description']}",
            allowed_tools=["Read", "Write", "Edit", "Bash(npm *,git *)"],
            max_turns=20
        )


async def run_multi_agent(task: str):
    """运行多 Agent 编排"""
    # 1. 分解任务
    plan = await orchestrator(task)
    subtasks = plan.get("subtasks", [])

    # 2. 并行执行
    results = await asyncio.gather(*[
        worker(st) for st in subtasks
    ])

    # 3. 汇总结果
    async with ClaudeAgentClient() as client:
        summary = await client.query(
            f"Synthesize these results into a final report: {results}"
        )
        return summary
```

## 安全执行模式

```python
import subprocess
import json


def safe_claude_execute(prompt: str, project_dir: str) -> dict:
    """安全执行 Claude Code 任务的封装"""
    # 1. Stash 当前改动
    subprocess.run(["git", "stash"], cwd=project_dir, check=True)

    try:
        # 2. 执行任务
        result = subprocess.run(
            [
                "claude",
                "-p", prompt,
                "--output-format", "json",
                "--max-turns", "30",
                "--allowedTools", "Read,Write,Edit,Bash(git *,npm *)",
                "--deniedTools", "Bash(sudo *,rm *,git push --force *)"
            ],
            capture_output=True,
            text=True,
            timeout=600,
            cwd=project_dir
        )

        output = json.loads(result.stdout)

        # 3. 验证
        if output.get("status") == "error":
            # 回滚
            subprocess.run(
                ["git", "checkout", "--", "."],
                cwd=project_dir, check=True
            )
            subprocess.run(["git", "stash", "pop"], cwd=project_dir)

        return output

    except Exception as e:
        # 出错回滚
        subprocess.run(
            ["git", "checkout", "--", "."],
            cwd=project_dir, check=True
        )
        subprocess.run(["git", "stash", "pop"], cwd=project_dir)
        raise e
```

## 监控与日志

```python
import logging
import time
from claude_agent_sdk import ClaudeAgentClient


logging.basicConfig(level=logging.INFO)
logger = logging.getLogger("claude-automation")


async def monitored_query(prompt: str, **kwargs):
    """带监控的查询执行"""
    start = time.time()
    logger.info(f"Starting task: {prompt[:100]}...")

    try:
        async with ClaudeAgentClient() as client:
            result = await client.query(prompt, **kwargs)

            elapsed = time.time() - start
            logger.info(
                f"Task completed | "
                f"Duration: {elapsed:.1f}s | "
                f"Tokens: {result.total_tokens} | "
                f"Turns: {result.total_turns} | "
                f"Model: {result.model}"
            )

            return result

    except Exception as e:
        elapsed = time.time() - start
        logger.error(f"Task failed after {elapsed:.1f}s: {e}")
        raise
```
