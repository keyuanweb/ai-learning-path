# 案例 4：企业数据分析平台 — 开源参考架构

## 背景

**项目**：[langgraph-multi-agent-analytics](https://github.com/zhang-juntai/langgraph-multi-agent-analytics)
**定位**：生产级 LangGraph 多 Agent 数据分析工作流平台
**目标**：超越"问答 + 工具调用"的简单 Agent，构建企业数据分析的完整闭环

## 为什么要超越"问答 + 工具调用"

传统数据分析 Agent 的六大痛点：

| 痛点 | 说明 | 后果 |
|------|------|------|
| **口径不稳定** | 同一个问题不同时间得到不同 SQL | 数据不一致 |
| **权限不可控** | Agent 缺乏细粒度权限 | 数据泄露风险 |
| **长任务上下文膨胀** | 多轮查询后上下文溢出 | 稳定性下降 |
| **人工介入后不可恢复** | 中断后无法从断点继续 | 工作流断裂 |
| **结果缺少验证** | 生成的 SQL/报告没有质量检查 | 错误决策 |
| **无审计追踪** | 谁在什么时候查了什么 | 合规风险 |

## 核心模块设计

### 1. 语义治理层（Semantic Layer）

确保数据口径一致性的关键。

```python
class SemanticLayer:
    """定义指标、维度、时间的标准语义"""

    metrics = {
        "revenue": {
            "name": "营收",
            "definition": "扣除退款后的实际收入",
            "sql": "SUM(amount) WHERE status != 'refunded'",
            "owner": "finance_team"
        },
        "dau": {
            "name": "日活跃用户",
            "definition": "当日至少有一次有效操作的去重用户数",
            "sql": "COUNT(DISTINCT user_id) WHERE actions > 0",
            "owner": "product_team"
        }
    }

    dimensions = {
        "country": {"table": "users", "column": "country_code"},
        "platform": {"table": "events", "column": "platform"},
    }

    def resolve_metric(self, name: str) -> str:
        """将用户说的'营收'转换为标准 SQL 定义"""
        if name in self.metrics:
            return self.metrics[name]["sql"]
        raise ValueError(f"未定义的指标: {name}")
```

### 2. 权限控制层（Row/Column-Level）

```python
class PermissionGate:
    """细粒度的数据和操作权限控制"""

    def __init__(self, user_id: str, role: str):
        self.user_id = user_id
        self.role = role

    def check_table_access(self, table: str) -> bool:
        """检查用户是否可以访问该表"""
        permissions = {
            "analyst": ["events", "users", "orders"],
            "manager": ["events", "users", "orders", "revenue", "costs"],
            "admin": ["*"]
        }
        allowed = permissions.get(self.role, [])
        return table in allowed or "*" in allowed

    def apply_row_filter(self, table: str, base_query: str) -> str:
        """为查询添加行级过滤（如销售只能看自己的区域）"""
        if table == "orders" and self.role == "analyst":
            return f"{base_query} AND region = '{self.user_region}'"
        return base_query

    def check_column_access(self, table: str, columns: list[str]) -> bool:
        """检查是否允许访问这些列"""
        pii_columns = {"email", "phone", "ssn", "credit_card"}
        if self.role != "admin":
            forbidden = [c for c in columns if c in pii_columns]
            if forbidden:
                return False
        return True
```

### 3. SQL 执行风控层

```python
from langgraph.types import interrupt

class QueryGuard:
    """SQL 查询的风险控制"""

    MAX_SCAN_GB = 500
    MAX_RUNTIME_MINUTES = 30
    MAX_COST_USD = 50

    @staticmethod
    def lint(sql: str) -> list[str]:
        """SQL 静态检查"""
        issues = []

        # 检查是否只读
        forbidden = ["INSERT", "UPDATE", "DELETE", "DROP", "ALTER",
                      "CREATE", "TRUNCATE", "GRANT", "REVOKE"]
        sql_upper = sql.upper()
        for kw in forbidden:
            if kw in sql_upper:
                issues.append(f"禁止的操作：{kw}")

        # 检查是否有 LIMIT（防止全表扫描）
        if "LIMIT" not in sql_upper and "SELECT" in sql_upper:
            issues.append("缺少 LIMIT 子句，建议添加")

        return issues

    @staticmethod
    def estimate_cost(sql: str) -> dict:
        """预估查询成本"""
        # EXPLAIN 获取查询计划
        plan = explain_query(sql)
        return {
            "scan_gb": plan.get("scan_bytes", 0) / 1e9,
            "estimated_runtime_ms": plan.get("runtime_ms", 0),
            "estimated_cost_usd": plan.get("cost_usd", 0)
        }

    @staticmethod
    def approve_high_cost(cost: dict) -> bool:
        """高成本查询需要人工审批"""
        if cost["estimated_cost_usd"] > QueryGuard.MAX_COST_USD:
            return interrupt(
                f"查询预估成本 ${cost['estimated_cost_usd']:.2f}，超过上限 ${QueryGuard.MAX_COST_USD}。"
                f"预估扫描 {cost['scan_gb']:.1f}GB。是否继续？"
            )
        return True
```

### 4. 结果验证层

```python
class ResultValidator:
    """查询结果的独立验证"""

    @staticmethod
    def check_freshness(result: pd.DataFrame, table: str) -> bool:
        """检查数据新鲜度"""
        if table == "events":
            max_ts = result["event_time"].max()
            hours_behind = (datetime.now() - max_ts).total_seconds() / 3600
            if hours_behind > 24:
                return False  # 超过 24 小时的旧数据
        return True

    @staticmethod
    def check_anomaly(result: pd.DataFrame, metric: str) -> bool:
        """检查结果是否异常（如营收突然为 0）"""
        if metric == "revenue" and result["revenue"].sum() == 0:
            return False  # 营收为 0 可能不正常
        return True

    @staticmethod
    def check_consistency(result: pd.DataFrame, expected_schema: dict) -> bool:
        """检查返回结构是否与预期一致"""
        actual_cols = set(result.columns)
        expected_cols = set(expected_schema.keys())
        return actual_cols == expected_cols
```

## Graph 编排实现

```python
def build_enterprise_analytics_graph() -> StateGraph:
    """构建完整的企业分析工作流"""

    builder = StateGraph(AnalyticsState)

    # 注册所有节点
    builder.add_node("planner", planner_node)
    builder.add_node("memory_hydrate", memory_hydrate_node)
    builder.add_node("clarification_gate", clarification_gate_node)
    builder.add_node("approval_gate", approval_gate_node)
    builder.add_node("prepare_query", prepare_query_node)
    builder.add_node("lint", lint_node)
    builder.add_node("execute", execute_node)
    builder.add_node("validate", validate_node)
    builder.add_node("report", report_node)
    builder.add_node("persist", persist_node)
    builder.add_node("checkpoint", checkpoint_node)
    builder.add_node("memory_persist", memory_persist_node)
    builder.add_node("alerts", alerts_node)

    # 连线
    builder.add_edge(START, "planner")
    builder.add_edge("planner", "memory_hydrate")
    builder.add_edge("memory_hydrate", "clarification_gate")
    # clarification 不明确 → 返回 planner 重新理解
    builder.add_conditional_edges("clarification_gate", clarification_router, {
        "clear": "approval_gate",
        "unclear": "planner"
    })
    builder.add_conditional_edges("approval_gate", approval_router, {
        "approved": "prepare_query",
        "rejected": END
    })
    builder.add_edge("prepare_query", "lint")
    builder.add_conditional_edges("lint", lint_router, {
        "pass": "execute",
        "fail": "prepare_query"  # 返回修改 SQL
    })
    builder.add_conditional_edges("execute", execute_router, {
        "success": "validate",
        "error": "alerts",
        "timeout": "alerts"
    })
    builder.add_conditional_edges("validate", validate_router, {
        "pass": "report",
        "fail": "alerts"
    })
    builder.add_edge("report", "persist")
    builder.add_edge("persist", "checkpoint")
    builder.add_edge("checkpoint", "memory_persist")
    builder.add_edge("memory_persist", END)
    builder.add_edge("alerts", END)

    return builder.compile(
        checkpointer=PostgresSaver.from_conn_string(os.environ["DATABASE_URL"]),
        interrupt_before=["approval_gate"]
    )
```

## 六大闭环总结

```
语义治理     →  统一指标/维度定义，消除口径歧义
权限控制     →  表级 + 行级 + 列级 + 操作级，四级权限
执行风控     →  Lint + Cost + Approve，高风险查询必须审批
结果验证     →  Freshness + Anomaly + Consistency，三重检查
断点恢复     →  Checkpoint + interrupt/resume，人工介入后可继续
可观测性     →  Audit + Metrics + Tracing + Alerts，全链路追踪
```

## 可复用的设计模式

1. **语义层**：任何涉及"指标"的 Agent 都需要语义层的口径治理
2. **权限四级**：表/行/列/操作的分层权限模型
3. **Lint → Execute → Validate**：代码生成 Agent 的标准安全流水线
4. **Clarification Gate**：执行前确认 Agent 是否真的理解了用户意图
5. **Checkpoint 全链路**：每个关键步骤后保存断点
